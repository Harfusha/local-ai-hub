+def receipt_verifier(task_store: TaskStore) -> VerificationStore:
+    verifier = getattr(task_store, "verification_store", None)
+    if verifier is None:
+        verifier = VerificationStore(
+            task_store.state_store,
+            task_store=task_store,
+            artifact_store=ArtifactStore(task_store.state_store.db_path.parent / "artifacts"),
+        )
+        task_store.verification_store = verifier
+    return verifier

+def passed_test_receipt(task_store: TaskStore, task_id: str, criterion: str, revision: str = "rev-1"):

+def test_task_transition_runs_verify_completion_with_valid_receipt(store: TaskStore):
+    task = task_ready_for_completion(store, "task-valid-receipt")
+    verifier = receipt_verifier(store)
+    verifier.record(passed_test_receipt(store, task.task_id, "tests"))
+    store._verify_completion(task, current_revision="rev-1")
+    completed = store.transition(
+        task.task_id, TaskStatus.COMPLETED, reason="verified receipt accepted", actor="agent",
+        idempotency_key="task-valid-receipt-complete",
+    )
+    assert completed.status is TaskStatus.COMPLETED
