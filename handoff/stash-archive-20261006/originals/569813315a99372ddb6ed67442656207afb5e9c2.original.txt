"""Build hash checks and status delta for T01 owner transfers."""

from __future__ import annotations

import csv
import hashlib
import json
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
CLAIMS = ROOT / "handoff/T01-root-transfer-claims-20261001-v2.json"
BASE_CSV = ROOT / "handoff/T01-operation-evidence-review-20261001-v1.csv"
OUT_CSV = ROOT / "handoff/T01-root-transfer-delta-20261001-v2.csv"
OUT_JSON = ROOT / "handoff/T01-root-transfer-delta-20261001-v2.json"
MERGED_CSV = ROOT / "handoff/T01-operation-evidence-review-20261001-v2.csv"


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest().upper()


def verify_artifact(item: dict[str, str] | None) -> dict[str, str]:
    if not item:
        return {"path": "", "claimed_sha256": "", "actual_sha256": "", "sha_status": "missing"}
    path = Path(item["path"])
    claimed = item.get("sha256_claimed", "") or item.get("sha256_observed_at_delta_build", "")
    if not path.is_file():
        return {"path": str(path), "claimed_sha256": claimed, "actual_sha256": "", "sha_status": "file_missing"}
    actual = sha256(path)
    return {
        "path": str(path),
        "claimed_sha256": claimed,
        "actual_sha256": actual,
        "sha_status": "matches" if claimed and claimed.upper() == actual else "no_claimed_sha" if not claimed else "mismatch",
    }


claims = json.loads(CLAIMS.read_text(encoding="utf-8"))
base_path = ROOT / claims["base_map"]["path"]
base_actual = sha256(base_path) if base_path.is_file() else ""
base_csv_actual = sha256(BASE_CSV) if BASE_CSV.is_file() else ""
records = []
for claim in claims["records"]:
    row = dict(claim)
    row["source_verification"] = [verify_artifact(source) for source in claim.get("source_files", [])]
    row["patch_verification"] = verify_artifact(claim.get("patch"))
    row["report_verification"] = verify_artifact(claim.get("report"))
    receipt = claim.get("receipt")
    row["receipt_verification"] = verify_artifact(receipt)
    row["receipt_sha_status"] = row["receipt_verification"]["sha_status"]
    row["source_run_binding_status"] = claim.get("source_run_binding_status", "not_transferred")
    row["source_sha_status"] = (
        "all_claimed_pins_match"
        if row["source_verification"] and all(item["sha_status"] == "matches" for item in row["source_verification"])
        else "partial_or_unbound"
        if row["source_verification"]
        else "not_supplied"
    )
    row["report_sha_status"] = row["report_verification"]["sha_status"]
    row["patch_sha_status"] = row["patch_verification"]["sha_status"]
    row["root_green"] = bool(
        claim["scope"].startswith("root")
        and claim.get("receipt")
        and row["receipt_sha_status"] == "matches"
        and row["source_sha_status"] == "all_claimed_pins_match"
        and row["source_run_binding_status"] == "verified_for_this_run"
        and "failed" not in claim["classification"]
    )
    records.append(row)

columns = [
    "record_id", "owner", "scope", "operation_id", "operation", "previous_operation_status", "applies_to_operation_map", "selector",
    "classification", "result", "command", "source_sha_status", "source_run_binding_status", "receipt_sha_status",
    "report_sha_status", "patch_sha_status", "root_green", "evidence_ref", "notes",
]
with OUT_CSV.open("w", encoding="utf-8", newline="") as handle:
    writer = csv.DictWriter(handle, fieldnames=columns, extrasaction="ignore")
    writer.writeheader()
    writer.writerows(records)

with BASE_CSV.open(encoding="utf-8-sig", newline="") as handle:
    base_rows = list(csv.DictReader(handle))
map_fields = list(base_rows[0]) + [
    "latest_transfer_status",
    "latest_transfer_selector",
    "latest_transfer_result",
    "latest_transfer_source_sha256",
    "latest_transfer_source_binding",
    "latest_transfer_receipt_sha256",
    "latest_transfer_receipt_sha_status",
    "latest_transfer_evidence_ref",
    "latest_transfer_root_green",
]
operation_updates = 0
for delta in records:
    if not delta.get("operation_id") or not delta.get("applies_to_operation_map"):
        continue
    for base_row in base_rows:
        if base_row.get("operation_id") != delta["operation_id"]:
            continue
        source_hash = next(
            (item["claimed_sha256"] or item["actual_sha256"] for item in delta["source_verification"] if item["actual_sha256"]),
            "",
        )
        base_row.update(
            latest_transfer_status=delta["classification"],
            latest_transfer_selector=delta["selector"],
            latest_transfer_result=delta["result"],
            latest_transfer_source_sha256=source_hash,
            latest_transfer_source_binding=delta["source_run_binding_status"],
            latest_transfer_receipt_sha256=delta["receipt_verification"]["actual_sha256"],
            latest_transfer_receipt_sha_status=delta["receipt_sha_status"],
            latest_transfer_evidence_ref=delta["evidence_ref"],
            latest_transfer_root_green=str(delta["root_green"]).lower(),
        )
        operation_updates += 1

with MERGED_CSV.open("w", encoding="utf-8", newline="") as handle:
    writer = csv.DictWriter(handle, fieldnames=map_fields, extrasaction="ignore")
    writer.writeheader()
    writer.writerows(base_rows)

summary = {
    "schema_version": "t01-root-transfer-delta/v2",
    "generated_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
    "builder": str(Path(__file__).relative_to(ROOT).as_posix()),
    "builder_sha256": sha256(Path(__file__)),
    "claims": str(CLAIMS.relative_to(ROOT).as_posix()),
    "claims_sha256": sha256(CLAIMS),
    "base_map": claims["base_map"]["path"],
    "base_map_claimed_sha256": claims["base_map"]["sha256"],
    "base_map_actual_sha256": base_actual,
    "base_map_sha_status": "matches" if base_actual == claims["base_map"]["sha256"] else "mismatch_or_missing",
    "base_map_csv": claims["base_map"]["csv_path"],
    "base_map_csv_claimed_sha256": claims["base_map"]["csv_sha256"],
    "base_map_csv_actual_sha256": base_csv_actual,
    "base_map_csv_sha_status": "matches" if base_csv_actual == claims["base_map"]["csv_sha256"] else "mismatch_or_missing",
    "operation_map_deltas": [
        {
            "operation_id": row["operation_id"],
            "previous_status": row["previous_operation_status"],
            "new_status": row["classification"],
            "applied": row["applies_to_operation_map"],
        }
        for row in records
        if row.get("operation_id") and row.get("applies_to_operation_map")
    ],
    "output_csv": str(OUT_CSV.relative_to(ROOT).as_posix()),
    "output_csv_sha256": sha256(OUT_CSV),
    "merged_operation_map_csv": str(MERGED_CSV.relative_to(ROOT).as_posix()),
    "merged_operation_map_csv_sha256": sha256(MERGED_CSV),
    "merged_operation_map_rows": len(base_rows),
    "operation_map_rows_updated": operation_updates,
    "records": records,
    "counts": {
        "total": len(records),
        "root_scope": sum(row["scope"].startswith("root") for row in records),
        "candidate_scope": sum(row["scope"].startswith("candidate") for row in records),
        "root_green": sum(row["root_green"] for row in records),
        "receipt_sha_matches": sum(row["receipt_sha_status"] == "matches" for row in records),
        "source_pin_sets_complete": sum(row["source_sha_status"] == "all_claimed_pins_match" for row in records),
        "source_run_bindings_verified": sum(row["source_run_binding_status"] == "verified_for_this_run" for row in records),
        "classifications": dict(sorted(Counter(row["classification"] for row in records).items())),
    },
}
OUT_JSON.write_text(json.dumps(summary, indent=2, sort_keys=True) + "\n", encoding="utf-8")
print(json.dumps({
    "rows": summary["counts"]["total"],
    "root_green": summary["counts"]["root_green"],
    "receipt_sha_matches": summary["counts"]["receipt_sha_matches"],
    "base_map_sha_status": summary["base_map_sha_status"],
    "csv_sha256": summary["output_csv_sha256"],
    "merged_operation_map_sha256": summary["merged_operation_map_csv_sha256"],
    "json_sha256": sha256(OUT_JSON),
}, sort_keys=True))
