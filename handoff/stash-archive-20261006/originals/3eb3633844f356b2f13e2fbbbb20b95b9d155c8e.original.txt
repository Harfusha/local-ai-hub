+from __future__ import annotations
+
+from pathlib import Path
+
+from local_ai_hub.agent_identity import ScopeContext
+from local_ai_hub.agent_tasks import GoalContract, TaskCheckpoint, TaskState, TaskStatus
+from local_ai_hub.verification_rules import _task_project_root
+
+
+def test_auto_worktree_path_is_task_project_root_over_source_and_context(tmp_path: Path):
+    source_root = tmp_path / "source"
+    worktree_root = tmp_path / "task-worktree"
+    source_root.mkdir()
+    worktree_root.mkdir()
+    task = TaskState(
+        task_id="task-worktree-root",
+        status=TaskStatus.PLANNED,
+        contract=GoalContract(goal="bind task project root"),
+        context=ScopeContext(repository_id=str(source_root), task_id="task-worktree-root"),
+        checkpoint=TaskCheckpoint(
+            state_data={
+                "repo_root": str(source_root),
+                "worktree": {
+                    "repo_root": str(source_root),
+                    "worktree_path": str(worktree_root),
+                },
+            }
+        ),
+    )
+
+    assert _task_project_root(task) == worktree_root.resolve()
