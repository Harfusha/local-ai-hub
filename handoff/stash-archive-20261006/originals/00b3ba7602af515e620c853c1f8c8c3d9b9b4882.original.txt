+from dataclasses import replace
+
+from local_ai_hub.agent_events import AgentStateStore
+from local_ai_hub.agent_identity import AgentScope, ScopeContext
+from local_ai_hub.agent_tasks import GoalContract, TaskStore
+from local_ai_hub.agent_verification import VerificationReceipt, VerificationStore
+from local_ai_hub.artifacts import ArtifactStore
+
+
+def _receipt(task, artifacts, *, passed, observed_at):
+    command_id = f"cmd-{task.task_id}-{'pass' if passed else 'fail'}"
+    exit_code = 0 if passed else 1
+    proof = {
+        "kind": "work_command_verification",
+        "passed": passed,
+        "task_id": task.task_id,
+        "owner": task.owner,
+        "criterion": "tests",
+        "command_id": command_id,
+        "command_classification": "test",
+        "command": "python -m pytest -q",
+        "repository_revision": "rev-1",
+        "mutation_generation": 0,
+        "exit_code": exit_code,
+        "cached": False,
+        "result": {"success": passed, "exit_code": exit_code},
+    }
+    evidence_id = artifacts.put_json(proof, task.owner, "command_verification")
+    return replace(
+        VerificationReceipt.create(
+            task.task_id,
+            "tests",
+            passed=passed,
+            evidence_id=evidence_id,
+            command_id=command_id,
+            owner=task.owner,
+            repository_revision="rev-1",
+            mutation_generation=0,
+            evidence_type="test",
+        ),
+        observed_at=observed_at,
+    )
+

+def test_newer_failed_receipt_supersedes_older_pass(tmp_path):
+    state = AgentStateStore(tmp_path / "agent-state.sqlite3", enabled=True)
+    tasks = TaskStore(state)
+    artifacts = ArtifactStore(tmp_path / "artifacts")
+    verifier = VerificationStore(state, task_store=tasks, artifact_store=artifacts)
+    task = tasks.create(
+        GoalContract(
+            goal="Verify latest receipt wins",
+            acceptance_criteria=("tests",),
+            scope=AgentScope.TASK,
+        ),
+        ScopeContext(repository_id="repo-receipt-order", task_id="task-receipt-order"),
+        task_id="task-receipt-order",
+        actor="tenant-a",
+    )
+
+    verifier.record(_receipt(task, artifacts, passed=True, observed_at=10.0))
+    verifier.record(_receipt(task, artifacts, passed=False, observed_at=20.0))
+    completion = verifier.completion(task.task_id, current_revision="rev-1", now=30.0)
+
+    assert completion.complete is False
+    assert completion.unsatisfied_criteria == ("tests",)
+

+def test_newer_failed_receipt_stays_authoritative_after_verifier_recreation(tmp_path):
+    database = tmp_path / "agent-state.sqlite3"
+    artifact_path = tmp_path / "artifacts"
+    state = AgentStateStore(database, enabled=True)
+    tasks = TaskStore(state)
+    artifacts = ArtifactStore(artifact_path)
+    verifier = VerificationStore(state, task_store=tasks, artifact_store=artifacts)
+    task = tasks.create(
+        GoalContract(
+            goal="Preserve latest receipt across store reopen",
+            acceptance_criteria=("tests",),
+            scope=AgentScope.TASK,
+        ),
+        ScopeContext(repository_id="repo-receipt-reopen", task_id="task-receipt-reopen"),
+        task_id="task-receipt-reopen",
+        actor="tenant-a",
+    )
+    verifier.record(_receipt(task, artifacts, passed=True, observed_at=10.0))
+    verifier.record(_receipt(task, artifacts, passed=False, observed_at=20.0))
+
+    reopened_verifier = VerificationStore(
+        state,
+        task_store=tasks,
+        artifact_store=ArtifactStore(artifact_path),
+    )
+    completion = reopened_verifier.completion(
+        task.task_id,
+        current_revision="rev-1",
+        now=30.0,
+    )
+
+    assert completion.complete is False
+    assert completion.unsatisfied_criteria == ("tests",)
