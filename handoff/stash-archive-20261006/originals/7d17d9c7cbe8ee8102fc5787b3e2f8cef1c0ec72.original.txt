from __future__ import annotations

import importlib.util
from pathlib import Path
import sys


ROOT_SOURCE = Path(r"C:\Users\Adam\.local-ai-hub\src")
T06_SOURCE = Path(r"C:\Users\Adam\.codex\worktrees\t06-safe-commands\.local-ai-hub\src")
T06_TEST = Path(
    r"C:\Users\Adam\.codex\worktrees\t06-safe-commands\.local-ai-hub\tests\test_t06_t09_validation_receipts.py"
)
CANDIDATE_ROOT = Path(__file__).resolve().parents[1]
OVERRIDES = {
    "local_ai_hub.agent_verification": CANDIDATE_ROOT
    / "src"
    / "local_ai_hub"
    / "agent_verification.py",
    "local_ai_hub.verification_rules": CANDIDATE_ROOT
    / "src"
    / "local_ai_hub"
    / "verification_rules.py",
    "local_ai_hub.commands": T06_SOURCE / "local_ai_hub" / "commands.py",
}

if not ROOT_SOURCE.exists() or not T06_TEST.exists():
    raise SystemExit("root source or T06 integration test is unavailable")

sys.path.insert(0, str(ROOT_SOURCE))
import local_ai_hub

for module_name, path in OVERRIDES.items():
    spec = importlib.util.spec_from_file_location(module_name, path)
    if spec is None or spec.loader is None:
        raise SystemExit(f"overlay module could not be loaded: {module_name}")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    spec.loader.exec_module(module)

from local_ai_hub import app, artifacts, commands, http_server, project_config

print(

    app.__file__,
    artifacts.__file__,
    http_server.__file__,

)
