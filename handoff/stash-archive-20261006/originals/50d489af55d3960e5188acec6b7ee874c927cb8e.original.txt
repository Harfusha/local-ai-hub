from __future__ import annotations

import importlib.util
from pathlib import Path
import sys


ROOT_SOURCE = Path(r"C:\Users\Adam\.local-ai-hub\src")
CANDIDATE_ROOT = Path(__file__).resolve().parents[1]
TEST = CANDIDATE_ROOT / "tests" / "test_project_completion_root.py"

sys.path.insert(0, str(ROOT_SOURCE))
import local_ai_hub

for module_name in (
    "local_ai_hub.process_utils",
    "local_ai_hub.agent_verification",
    "local_ai_hub.verification_rules",
):
    candidate_path = CANDIDATE_ROOT / "src" / "local_ai_hub" / f"{module_name.rsplit('.', 1)[-1]}.py"
    spec = importlib.util.spec_from_file_location(module_name, candidate_path)
    if spec is None or spec.loader is None:
        raise SystemExit(f"candidate module could not be loaded: {module_name}")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    spec.loader.exec_module(module)

candidate_task_path = (
    CANDIDATE_ROOT / "src" / "local_ai_hub" / "agent_tasks.py"
)
candidate_task_spec = importlib.util.spec_from_file_location(
    "local_ai_hub._t09_candidate_agent_tasks", candidate_task_path
)
if candidate_task_spec is None or candidate_task_spec.loader is None:
    raise SystemExit("candidate task module could not be loaded")
candidate_task_module = importlib.util.module_from_spec(candidate_task_spec)
sys.modules[candidate_task_spec.name] = candidate_task_module
candidate_task_spec.loader.exec_module(candidate_task_module)

from local_ai_hub import agent_tasks, app, http_server, project_config

candidate_task_module.CompletionGateError = agent_tasks.CompletionGateError
candidate_task_module.InvalidTransitionError = agent_tasks.InvalidTransitionError
candidate_task_module.TaskStatus = agent_tasks.TaskStatus
agent_tasks.TaskStore.complete = candidate_task_module.TaskStore.complete
agent_tasks.TaskStore._verify_completion = (
    candidate_task_module.TaskStore._verify_completion
)

print(
    "root modules:", app.__file__, http_server.__file__, project_config.__file__,
    "candidate task method:", candidate_task_path,
    "candidate root adapter:", sys.modules["local_ai_hub.verification_rules"].__file__,
)

import pytest

raise SystemExit(
    pytest.main(
        [str(TEST), "-q", "--tb=short"]
    )
)
