import csv
from pathlib import Path

inventory = Path(__file__).with_name("T01-T27-http-post-route-inventory-20261003.csv")
with inventory.open(encoding="utf-8-sig", newline="") as stream:
    rows = list(csv.DictReader(stream))

assert len(rows) == 223
assert len({row["path"] for row in rows}) == 223
assert len({row["operation_id"] for row in rows}) == 223
assert all(row["method"] == "POST" and row["handler"] == "do_POST" and row["actions"] for row in rows)
assert all(row["owner_category"] and row["owner_evidence"] for row in rows)
assert sum(row["owner_category"] == "unassigned" for row in rows) == 195
assert all(row["owner_evidence"].startswith("No exact POST owner mapping") for row in rows if row["owner_category"] == "unassigned")
statuses = {}
for row in rows:
    statuses[row["body_schema_status"]] = statuses.get(row["body_schema_status"], 0) + 1
assert statuses == {
    "API_COVERAGE body fields present": 185,
    "source-observed body fields; requiredness, types, and completeness unverified": 33,
    "source confirms service consumes no request payload; generic JSON object gate remains": 4,
    "raw binary body; no JSON schema": 1,
}
assert all(row["body_schema_source"] for row in rows)
assert sum(not row["body_fields"] for row in rows) == 5
print("PASS: 223 POST rows; unique paths/op IDs; owner evidence on all rows; 195 explicitly unassigned; body status = 185 documented, 33 source-observed partial, 4 payload-unused, 1 raw binary.")
